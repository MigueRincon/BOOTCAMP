1. Definir Esquemas

Los esquemas definen la estructura de tus documentos en MongoDB

¿Qué es un Esquema?

Un esquema de Mongoose define la estructura, tipos de datos y validaciones para tus documentos en MongoDB. Es como una plantilla para tus datos.

Características clave:

  • Tipado: Define qué tipo de dato es cada campo
  • Validaciones: Reglas que deben cumplir los datos
  • Valores por defecto: Valores que se asignan automáticamente
  • Métodos personalizados: Funciones que puedes ejecutar

Ejemplo: Esquema de Usuarios

models/User.js 
const mongoose = require('mongoose');

const userSchema = new mongoose.Schema({
  nombre: {
    type: String,
    required: [true, 'El nombre es requerido'],
    trim: true,
    minlength: [2, 'El nombre debe tener al menos 2 caracteres'],
    maxlength: [50, 'El nombre no puede exceder 50 caracteres']
  },
  
  email: {
    type: String,
    required: [true, 'El email es requerido'],
    unique: true,
    lowercase: true,
    match: [/^\w+([\.-]?\w+)*@\w+([\.-]?\w+)*(\.\w{2,3})+$/, 'Email inválido']
  },
  
  edad: {
    type: Number,
    min: [18, 'Debes ser mayor de 18'],
    max: [120, 'Edad no válida']
  },
  
  ciudad: {
    type: String,
    default: 'No especificada'
  },
  
  telefono: {
    type: String,
    sparse: true
  },
  
  activo: {
    type: Boolean,
    default: true
  },
  
  fechaCreacion: {
    type: Date,
    default: Date.now
  },
  
  rol: {
    type: String,
    enum: ['usuario', 'admin', 'moderador'],
    default: 'usuario'
  }
}, { timestamps: true });

module.exports = mongoose.model('User', userSchema);

📌 Nota: El parámetro { timestamps: true } agrega automáticamente createdAt y updatedAt a cada documento.

Ejemplo: Esquema de Productos

models/Product.js 
const mongoose = require('mongoose');

const productSchema = new mongoose.Schema({
  nombre: {
    type: String,
    required: true,
    trim: true
  },
  
  descripcion: {
    type: String,
    required: true
  },
  
  precio: {
    type: Number,
    required: true,
    min: [0, 'El precio no puede ser negativo']
  },
  
  cantidad: {
    type: Number,
    required: true,
    default: 0
  },
  
  categoria: {
    type: String,
    required: true,
    enum: ['Electrónica', 'Ropa', 'Alimentos', 'Otros']
  },
  
  disponible: {
    type: Boolean,
    default: true
  },
  
  sku: {
    type: String,
    unique: true,
    sparse: true
  }
}, { timestamps: true });

module.exports = mongoose.model('Product', productSchema);

2. Estructura del Proyecto

Cómo organizar tu proyecto Express + MongoDB

Carpetas recomendadas

proyecto/ │ ├── models/ # Esquemas de Mongoose │ ├── User.js │ ├── Product.js │ └── Order.js │ ├── routes/ # Rutas/Endpoints │ ├── userRoutes.js │ ├── productRoutes.js │ └── orderRoutes.js │ ├── controllers/ # Lógica de negocio │ ├── userController.js │ ├── productController.js │ └── orderController.js │ ├── middleware/ # Middleware personalizado │ └── errorHandler.js │ ├── config/ # Configuración │ └── db.js │ ├── app.js # Configuración principal ├── server.js # Iniciar servidor └── .env # Variables de entorno

Archivo de Conexión (config/db.js)

config/db.js 
const mongoose = require('mongoose');

const conectarDB = async () => {
  try {
    const conexion = await mongoose.connect(process.env.MONGODB_URI || 'mongodb://localhost:27017/miapp', {
      useNewUrlParser: true,
      useUnifiedTopology: true
    });
    
    console.log('✅ MongoDB conectada:', conexion.connection.host);
    return conexion;
  } catch (error) {
    console.error('❌ Error al conectar MongoDB:', error.message);
    process.exit(1);
  }
};

module.exports = conectarDB;

3. Definir Rutas

Endpoints para acceder a los datos

¿Qué son las Rutas?

Las rutas (routes) definen los endpoints de tu API. Cada ruta mapea una URL a un controlador que manejará la lógica.

Rutas de Usuarios

routes/userRoutes.js 
const express = require('express');
const router = express.Router();
const {
  obtenerUsuarios,
  obtenerUsuarioPorId,
  crearUsuario,
  actualizarUsuario,
  eliminarUsuario
} = require('../controllers/userController');

// GET - Obtener todos los usuarios
router.get('/', obtenerUsuarios);

// GET - Obtener un usuario por ID
router.get('/:id', obtenerUsuarioPorId);

// POST - Crear un nuevo usuario
router.post('/', crearUsuario);

// PUT - Actualizar un usuario
router.put('/:id', actualizarUsuario);

// DELETE - Eliminar un usuario
router.delete('/:id', eliminarUsuario);

module.exports = router;

💡 Convención: Usa métodos HTTP estándar: GET (lectura), POST (creación), PUT (actualización), DELETE (eliminación).

Registrar Rutas en app.js

app.js 
const express = require('express');
const conectarDB = require('./config/db');
const userRoutes = require('./routes/userRoutes');
const productRoutes = require('./routes/productRoutes');

const app = express();

// Conectar a MongoDB
conectarDB();

// Middleware
app.use(express.json());
app.use(express.urlencoded({ extended: true }));

// Registrar rutas
app.use('/api/users', userRoutes);
app.use('/api/products', productRoutes);

// Ruta de prueba
app.get('/', (req, res) => {
  res.json({ mensaje: '¡API funcionando!' });
});

module.exports = app;

4. Crear Controlador

Lógica de negocio y manejo de peticiones

¿Qué es un Controlador?

Un controlador contiene la lógica que se ejecuta cuando alguien accede a una ruta. Aquí consultas la base de datos, procesas datos y envías respuestas.

Controlador de Usuarios

controllers/userController.js 
const User = require('../models/User');

// GET - Obtener todos los usuarios
exports.obtenerUsuarios = async (req, res) => {
  try {
    // Consulta a MongoDB
    const usuarios = await User.find();
    
    res.status(200).json({
      exitoso: true,
      cantidad: usuarios.length,
      datos: usuarios
    });
  } catch (error) {
    res.status(500).json({
      exitoso: false,
      mensaje: 'Error al obtener usuarios',
      error: error.message
    });
  }
};

// GET - Obtener un usuario por ID
exports.obtenerUsuarioPorId = async (req, res) => {
  try {
    const { id } = req.params;
    
    const usuario = await User.findById(id);
    
    if (!usuario) {
      return res.status(404).json({
        exitoso: false,
        mensaje: 'Usuario no encontrado'
      });
    }
    
    res.status(200).json({
      exitoso: true,
      datos: usuario
    });
  } catch (error) {
    res.status(500).json({
      exitoso: false,
      mensaje: 'Error al obtener usuario',
      error: error.message
    });
  }
};

// POST - Crear un nuevo usuario
exports.crearUsuario = async (req, res) => {
  try {
    const { nombre, email, edad, ciudad, rol } = req.body;
    
    // Crear nuevo documento
    const nuevoUsuario = new User({
      nombre,
      email,
      edad,
      ciudad,
      rol
    });
    
    // Guardar en MongoDB
    const usuarioGuardado = await nuevoUsuario.save();
    
    res.status(201).json({
      exitoso: true,
      mensaje: 'Usuario creado exitosamente',
      datos: usuarioGuardado
    });
  } catch (error) {
    res.status(400).json({
      exitoso: false,
      mensaje: 'Error al crear usuario',
      error: error.message
    });
  }
};

// PUT - Actualizar un usuario
exports.actualizarUsuario = async (req, res) => {
  try {
    const { id } = req.params;
    const datosActualizados = req.body;
    
    const usuarioActualizado = await User.findByIdAndUpdate(
      id,
      datosActualizados,
      { new: true, runValidators: true }
    );
    
    if (!usuarioActualizado) {
      return res.status(404).json({
        exitoso: false,
        mensaje: 'Usuario no encontrado'
      });
    }
    
    res.status(200).json({
      exitoso: true,
      mensaje: 'Usuario actualizado',
      datos: usuarioActualizado
    });
  } catch (error) {
    res.status(400).json({
      exitoso: false,
      mensaje: 'Error al actualizar usuario',
      error: error.message
    });
  }
};

// DELETE - Eliminar un usuario
exports.eliminarUsuario = async (req, res) => {
  try {
    const { id } = req.params;
    
    const usuarioEliminado = await User.findByIdAndDelete(id);
    
    if (!usuarioEliminado) {
      return res.status(404).json({
        exitoso: false,
        mensaje: 'Usuario no encontrado'
      });
    }
    
    res.status(200).json({

    });
  } catch (error) {
    res.status(500).json({
      exitoso: false,
      mensaje: 'Error al eliminar usuario',
      error: error.message
    });
  }
};

5. Consultas MongoDB

Métodos principales para buscar y manipular datos

Métodos de Consulta Principales

Consultas Básicas 
// Obtener TODOS los documentos
const usuarios = await User.find();

// Obtener por ID
const usuario = await User.findById(id);

// Obtener por un campo específico
const usuario = await User.findOne({ email: 'juan@email.com' });

// Contar documentos
const cantidad = await User.countDocuments();

// Obtener solo ciertos campos
const usuarios = await User.find().select('nombre email edad');

// Excluir ciertos campos
const usuarios = await User.find().select('-__v -fechaCreacion');
Consultas Avanzadas 
// Filtrar por condiciones (usuarios mayores de 30)
const usuarios = await User.find({ edad: { $gte: 30 } });

// Filtrar por múltiples condiciones (AND)
const usuarios = await User.find({ 
  edad: { $gte: 25 },
  ciudad: 'Bogotá'
});

// Filtrar por OR
const usuarios = await User.find({
  $or: [
    { rol: 'admin' },
    { rol: 'moderador' }
  ]
});

// Ordenar resultados (1 = ascendente, -1 = descendente)
const usuarios = await User.find().sort({ nombre: 1 });
const usuarios = await User.find().sort({ fechaCreacion: -1 });

// Limitar resultados
const usuarios = await User.find().limit(10);

// Saltar documentos (paginación)
const pagina = 2;
const porPagina = 10;
const usuarios = await User.find()
  .skip((pagina - 1) * porPagina)
  .limit(porPagina);

// Buscar texto
const usuarios = await User.find(
  { $text: { $search: 'juan' } }
);
Con Parámetros de Query 
// Controlador con búsqueda y filtro flexible
exports.obtenerProductos = async (req, res) => {
  try {
    // ?categoria=Electrónica&minPrecio=100&maxPrecio=5000&pagina=1
    const { categoria, minPrecio, maxPrecio, pagina = 1, limite = 10 } = req.query;
    
    // Construir filtro dinámico
    let filtro = {};
    
    if (categoria) {
      filtro.categoria = categoria;
    }
    
    if (minPrecio || maxPrecio) {
      filtro.precio = {};
      if (minPrecio) filtro.precio.$gte = Number(minPrecio);
      if (maxPrecio) filtro.precio.$lte = Number(maxPrecio);
    }
    
    if (req.query.disponible === 'true') {
      filtro.disponible = true;
    }
    
    // Ejecutar consulta con paginación
    const productos = await Product.find(filtro)
      .limit(Number(limite))
      .skip((Number(pagina) - 1) * Number(limite))
      .sort({ nombre: 1 });
    
    const total = await Product.countDocuments(filtro);
    
    res.json({
      exitoso: true,
      datos: productos,
      paginacion: {
        total,
        pagina: Number(pagina),
        paginas: Math.ceil(total / Number(limite))
      }
    });
  } catch (error) {
    res.status(500).json({ exitoso: false, error: error.message });
  }
};

Operadores de Comparación

Operadores MongoDB 
// Igual (por defecto)
{ edad: 25 }

// Mayor que ($gt) y Mayor o igual ($gte)
{ edad: { $gt: 25 } }      // edad > 25
{ edad: { $gte: 25 } }     // edad >= 25

// Menor que ($lt) y Menor o igual ($lte)
{ edad: { $lt: 65 } }      // edad < 65
{ edad: { $lte: 65 } }     // edad <= 65

// No igual ($ne)
{ rol: { $ne: 'usuario' } }

// En una lista ($in)
{ estado: { $in: ['activo', 'pendiente'] } }

// No en una lista ($nin)
{ estado: { $nin: ['eliminado', 'archivado'] } }

// Existe o no existe ($exists)
{ telefono: { $exists: true } }

// Coincide con expresión regular ($regex)
{ nombre: { $regex: /juan/i } }  // 'i' = case insensitive

Fetch API 
// GET - Obtener usuarios
async function obtenerUsuarios() {
  try {
    const response = await fetch('http://localhost:3000/api/users');
    const data = await response.json();
    console.log(data);
  } catch (error) {
    console.error('Error:', error);
  }
}

// POST - Crear usuario
async function crearUsuario(nuevoUsuario) {
  try {
    const response = await fetch('http://localhost:3000/api/users', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify(nuevoUsuario)
    });
    const data = await response.json();
    console.log('Usuario creado:', data);
  } catch (error) {
    console.error('Error:', error);
  }
}

// PUT - Actualizar usuario
async function actualizarUsuario(id, datosActualizados) {
  try {
    const response = await fetch(`http://localhost:3000/api/users/${id}`, {
      method: 'PUT',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify(datosActualizados)
    });
    const data = await response.json();
    console.log('Usuario actualizado:', data);
  } catch (error) {
    console.error('Error:', error);
  }
}

// DELETE - Eliminar usuario
async function eliminarUsuario(id) {
  try {
    const response = await fetch(`http://localhost:3000/api/users/${id}`, {
      method: 'DELETE'
    });
    const data = await response.json();
    console.log('Usuario eliminado:', data);
  } catch (error) {
    console.error('Error:', error);
  }
}

5. URLs de tu API

Método URL Descripción
GET /api/users Todos los usuarios
GET /api/users/:id Usuario por ID
POST /api/users Crear usuario
PUT /api/users/:id Actualizar usuario
DELETE /api/users/:id Eliminar usuario

⚠️ IMPORTANTE: Asegúrate de que MongoDB esté ejecutándose en localhost:27017 o usa MongoDB Atlas con la URI en el .env.

Flujo Completo de una Petición

1️⃣ Cliente hace petición GET /api/users 2️⃣ Express recibe en app.js app.use('/api/users', userRoutes) 3️⃣ Router busca la ruta router.get('/', obtenerUsuarios) 4️⃣ Se ejecuta el controlador exports.obtenerUsuarios = async (req, res) => { ... } 5️⃣ Controlador consulta MongoDB const usuarios = await User.find() 6️⃣ MongoDB retorna datos [{ _id, nombre, email, ... }, ...] 7️⃣ Controlador envía respuesta res.json({ exitoso: true, datos: usuarios }) 8️⃣ Cliente recibe JSON { exitoso: true, datos: [...] }
